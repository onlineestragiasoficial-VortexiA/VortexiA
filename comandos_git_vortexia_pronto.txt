# Inicialize um repositório Git
git init

# Adicione todos os arquivos ao repositório
git add .

# Faça o primeiro commit
git commit -m "Primeiro commit do Frontend"

# Adicione o repositório remoto
git remote add origin https://github.com/onlineestragiasoficial-VortexiA/vortexia-frontend.git

# Envie os arquivos para o repositório
git push -u origin master
